local IGNORE_FILE = "NullHub_ServerHop.txt"
local IGNORE_HOUR = 3600
local function getIgnored()
    local list = {}
    if not isfile or not isfile(IGNORE_FILE) then return list end
    local ok, content = pcall(readfile, IGNORE_FILE)
    if not ok or not content then return list end
    local now = os.time()
    for line in content:gmatch("[^\n]+") do
        local id, ts = line:match("^([^|]+)|(%d+)$")
        ts = tonumber(ts) or 0
        if id and (now - ts) < IGNORE_HOUR then list[id] = ts end
    end
    return list
end
local function saveIgnored(list)
    if not writefile then return end
    local lines = {}
    for id, ts in pairs(list) do table.insert(lines, id .. "|" .. tostring(ts)) end
    pcall(writefile, IGNORE_FILE, table.concat(lines, "\n"))
end
local Ignored = getIgnored()
local function markIgnored(jobId) Ignored[jobId] = os.time(); saveIgnored(Ignored) end
local function clearIgnored() Ignored = {}; if writefile then pcall(writefile, IGNORE_FILE, "") end end

local function rawRequest(url, method, body)
    local headers = { ["Accept"] = "application/json", ["User-Agent"] = "Roblox/WinInet" }
    if method == "POST" then headers["Content-Type"] = "application/json" end
    local tryFns = {}
    if syn    and syn.request    then table.insert(tryFns, syn.request)    end
    if http   and http.request   then table.insert(tryFns, http.request)   end
    if request                   then table.insert(tryFns, request)        end
    if http_request              then table.insert(tryFns, http_request)   end
    if fluxus and fluxus.request then table.insert(tryFns, fluxus.request) end
    local req = { Url = url, Method = method or "GET", Headers = headers }
    if body then req.Body = body end
    for _, fn in ipairs(tryFns) do
        local ok, res = pcall(fn, req)
        if ok and type(res) == "table" then
            local b = res.Body or res.body or ""
            local code = res.StatusCode or res.statusCode or 200
            if b ~= "" then return b, code end
        end
    end
    if not method or method == "GET" then
        local ok, b = pcall(function() return game:HttpGet(url, true) end)
        if ok and type(b) == "string" and b ~= "" then return b, 200 end
        local ok2, b2 = pcall(function() return HttpService:GetAsync(url, true) end)
        if ok2 and type(b2) == "string" and b2 ~= "" then return b2, 200 end
    end
    return nil, 0
end

local function httpGet(url, maxRetries)
    maxRetries = maxRetries or 4
    local proxyUrl = url:gsub("games%.roblox%.com", "games.roproxy.com")
    local delay = 1.5
    for attempt = 1, maxRetries do
        local body, code = rawRequest(url)
        if not body or code == 429 then body, code = rawRequest(proxyUrl) end
        if body and body ~= "" then
            local is429 = (code == 429)
            if not is429 then
                local lo = body:lower()
                if lo:find('"toomanyrequests"') or lo:find("too many requests") then is429 = true end
            end
            if is429 then
                if attempt < maxRetries then task.wait(delay); delay = delay * 2 end
            else
                return body, code
            end
        else
            if attempt < maxRetries then task.wait(delay); delay = delay * 2 end
        end
    end
    return nil, 429
end

local function parseServers(body)
    if not body or body == "" then return false end
    local ok, decoded = pcall(function() return HttpService:JSONDecode(body) end)
    if not ok or type(decoded) ~= "table" or decoded.errors then return false end
    local list = decoded.data
    if type(list) ~= "table" then return false end
    local existing = {}
    for _, ex in ipairs(S.servers) do existing[ex.id] = true end
    for _, srv in ipairs(list) do
        if type(srv) == "table" and type(srv.id) == "string" and srv.id ~= "" then
            local playing = tonumber(srv.playing) or 0
            local maxP    = tonumber(srv.maxPlayers) or 0
            local ping    = tonumber(srv.ping) or 0
            if not existing[srv.id] and (maxP == 0 or playing < maxP) then
                table.insert(S.servers, { id = srv.id, playing = playing, max = maxP, ping = ping })
                existing[srv.id] = true
                S.totalLoaded = S.totalLoaded + 1
            end
        end
    end
    local nc = decoded.nextPageCursor
    S.cursor = (type(nc) == "string" and nc ~= "") and nc or nil
    return true
end

local function buildURL(cursor)
    local url = ("https://games.roblox.com/v1/games/%d/servers/Public?limit=100&sortOrder=Asc"):format(PLACE_ID)
    if cursor and cursor ~= "" then url = url .. "&cursor=" .. HttpService:UrlEncode(cursor) end
    return url
end

local function loadAllServers()
    S.servers = {}; S.cursor = nil; S.totalLoaded = 0
    local pages = 0
    repeat
        local body = httpGet(buildURL(S.cursor))
        if not body then break end
        if not parseServers(body) then break end
        pages = pages + 1
        if S.cursor then task.wait(0.4) end
    until not S.cursor or pages >= 8
    return true
end

local function loadFriends()
    S.friends = {}
    local function decodeBody(body)
        if not body then return nil end
        local ok, d = pcall(function() return HttpService:JSONDecode(body) end)
        if ok and type(d) == "table" then return d end
        return nil
    end
    local body1 = rawRequest("https://friends.roblox.com/v1/users/" .. LP.UserId .. "/friends")
    if not body1 then showNotification("Friends", "Could not fetch friend list", 4) return false end
    local d1 = decodeBody(body1)
    if not d1 or type(d1.data) ~= "table" then
        showNotification("Friends", "Invalid friend list response", 4) return false
    end
    local friendList = d1.data
    if #friendList == 0 then showNotification("Friends", "You have no friends on Roblox", 4) return true end

    local ids, idToName = {}, {}
    for _, f in ipairs(friendList) do
        if type(f) == "table" and tonumber(f.id) then
            local uid = tonumber(f.id)
            table.insert(ids, uid)
            idToName[uid] = f.displayName or f.name or tostring(uid)
        end
    end

    local matched, total = 0, 0
    local function queryPresence(baseUrl)
        local chunkStart = 1
        while chunkStart <= #ids do
            local chunk = {}
            for j = chunkStart, math.min(chunkStart + 99, #ids) do
                table.insert(chunk, ids[j])
            end
            local payload = HttpService:JSONEncode({ userIds = chunk })
            local body2 = rawRequest(baseUrl, "POST", payload)
            local d2 = decodeBody(body2)
            if d2 and type(d2.userPresences) == "table" then
                for _, pres in ipairs(d2.userPresences) do
                    total = total + 1
                    local uid = tonumber(pres.userId)
                    local pType = tonumber(pres.userPresenceType) or 0
                    local placeA = tonumber(pres.placeId) or 0
                    local placeB = tonumber(pres.rootPlaceId) or 0
                    local jobId  = pres.gameId
                    local inPlace = (placeA == PLACE_ID) or (placeB == PLACE_ID)
                    local hasJob = type(jobId) == "string" and jobId ~= ""
                    if (pType == 2 or pType == 3) and inPlace and hasJob then
                        table.insert(S.friends, { name = idToName[uid] or tostring(uid), uid = uid, jobId = jobId })
                        matched = matched + 1
                    end
                end
            end
            chunkStart = chunkStart + 100
        end
    end

    queryPresence("https://presence.roblox.com/v1/presence/users")
    if total == 0 then queryPresence("https://presence.roproxy.com/v1/presence/users") end

    if total == 0 then showNotification("Friends", "Presence API unavailable", 5) return false end
    if matched == 0 then showNotification("Friends", ("0 of %d friends in this game"):format(#ids), 4)
    else showNotification("Friends", ("%d friend(s) in game"):format(matched), 3) end
    return true
end

local function filteredServers()
    local out = {}
    for _, srv in ipairs(S.servers) do
        local isFull = (srv.max > 0 and srv.playing >= srv.max)
        if not isFull and not Ignored[srv.id] then
            local keep = (S.filter == "all") or (S.filter == "one" and srv.playing == 1)
            if keep then table.insert(out, srv) end
        end
    end
    table.sort(out, function(a,b)
        if a.playing ~= b.playing then return a.playing < b.playing end
        return a.ping < b.ping
    end)
    return out
end

local function queueReExec()
    local url = Config.url
    if url == "" or not url then
        showNotification("Auto-Exec", "No hub URL set", 3)
        return
    end
    local code = string.format([[task.wait(2) pcall(function() loadstring(game:HttpGet(%q, true))() end)]], url)
    local queued = false
    for _, fn in ipairs({
        queue_on_teleport,
        syn and syn.queue_on_teleport,
        queueonteleport,
        fluxus and fluxus.queue_on_teleport,
    }) do
        if fn then
            local ok = pcall(fn, code)
            if ok then queued = true end
        end
    end
    if queued then
        showNotification("Auto-Exec", "Null Hub queued for next server", 2)
    end
end

local function hopTo(jobId)
    if S.hopping then return end
    S.hopping = true
    S.lastStatus = "joining..."
    if _G.__nullRefresh then _G.__nullRefresh() end
    markIgnored(jobId)
    if Config.autoExec and Config.url and Config.url ~= "" then
        queueReExec()
    end
    local ok = pcall(function() TeleportService:TeleportToPlaceInstance(PLACE_ID, jobId, LP) end)
    if not ok then S.hopping = false; S.lastStatus = "join failed"; if _G.__nullRefresh then _G.__nullRefresh() end end
end

local function hopToLowest()
    local list = filteredServers()
    if #list == 0 then
        local fallback = {}
        for _, srv in ipairs(S.servers) do
            if not (srv.max > 0 and srv.playing >= srv.max) and not Ignored[srv.id] then
                table.insert(fallback, srv)
            end
        end
        table.sort(fallback, function(a,b) return a.playing < b.playing end)
        if #fallback == 0 then S.lastStatus = "no servers"; if _G.__nullRefresh then _G.__nullRefresh() end; return end
        hopTo(fallback[1].id); return
    end
    hopTo(list[1].id)
end

    local body = httpGet(url, 2)
    if not body then showNotification("Search", "Request failed", 4) return end

end

local pages = {}

local currentTab = nil

local function switchTab(name)
    for n, p in pairs(pages) do
        p.Visible = false
        if tabBtns[n] then
            tabBtns[n].BackgroundColor3 = T.card
            tabBtns[n].TextColor3 = T.dim
        end
    end
    if pages[name] then
        pages[name].Visible = true
        if tabBtns[name] then
            tabBtns[name].BackgroundColor3 = T.white
            tabBtns[name].TextColor3 = T.bg
        end
        currentTab = name
    end
end

local function runSearch(query, page)
    if not query or query == "" then showNotification("Search", "Type something", 2) return end
    S.searchQuery = query
    if page == 1 then clearSearchResults() end
    S.searchPage = page
    local params = "q="..HttpService:UrlEncode(query).."&page="..page.."&max=10"
    if S.keylessOnly then params = params.."&mode=free" end
    showNotification("Search", "Searching \""..query.."\"", 2)
    fetchScripts(params, function(scripts, totalPages)
        if #scripts == 0 then showNotification("Search", "No results", 3) moreBtn.Visible=false return end
        local start = #S.searchResults
        for i,s in ipairs(scripts) do
            local n = normalizeSB(s)
            table.insert(S.searchResults, n)
            buildScriptCard(resultsHolder, n, start + i)
        end
        moreBtn.Visible = page < totalPages
        moreBtn.Text = "Load More  ("..page.."/"..totalPages..")"
        showNotification("Search", #scripts.." results", 2)
    end)
end

local function clearESP()

local function makeESPFor(plr)
    if plr == LP then return end
    local function setup(char)
        if not esp.enabled then return end
        task.wait(0.3)
        if not char or not char.Parent then return end
        if esp.highlights[plr] then return end

        local hl = Instance.new("Highlight")
        hl.FillTransparency = 1
        hl.OutlineColor = T.white
        hl.OutlineTransparency = 0
        hl.Adornee = char
        hl.Parent = gui
        esp.highlights[plr] = hl

        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        local head = char:FindFirstChild("Head") or root

        local bg = Instance.new("BillboardGui")
        bg.Name = "NullESP"
        bg.Size = UDim2.new(0, 180, 0, 40)
        bg.StudsOffset = Vector3.new(0, 3.2, 0)
        bg.AlwaysOnTop = true
        bg.Adornee = head
        bg.Parent = gui
        esp.labels[plr] = bg

        local nameL = Instance.new("TextLabel", bg)
        nameL.Size = UDim2.new(1, 0, 0, 16)
        nameL.BackgroundTransparency = 1
        nameL.Text = plr.Name
        nameL.TextColor3 = T.white
        nameL.TextStrokeTransparency = 0
        nameL.TextStrokeColor3 = Color3.new(0,0,0)
        nameL.TextSize = 13
        nameL.Font = F_BOLD

        local distL = Instance.new("TextLabel", bg)
        distL.Size = UDim2.new(1, 0, 0, 12)
        distL.Position = UDim2.new(0, 0, 0, 16)
        distL.BackgroundTransparency = 1
        distL.Text = ""
        distL.TextColor3 = T.dim
        distL.TextStrokeTransparency = 0.3
        distL.TextSize = 10
        distL.Font = F_MONO

        task.spawn(function()
            while esp.enabled and esp.labels[plr] == bg and char.Parent do
                local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
                if myRoot and root.Parent then
                    local d = math.floor((myRoot.Position - root.Position).Magnitude)
                    distL.Text = "[" .. d .. " studs]"
                end
                task.wait(0.15)
            end
        end)
    end
    if plr.Character then task.spawn(setup, plr.Character) end
    plr.CharacterAdded:Connect(function(c) if esp.enabled then task.spawn(setup, c) end end)
end

local function renderServerRow(srv, order)
    local row=mk("TextButton",{
        Size=UDim2.new(1,0,0,28), BackgroundColor3=T.card,
        BorderSizePixel=0, Text="", AutoButtonColor=false, LayoutOrder=order,
    },listFrame)
    Instance.new("UICorner",row).CornerRadius=UDim.new(0,3)
    mk("TextLabel",{Size=UDim2.new(0,56,1,0),Position=UDim2.fromOffset(8,0),
        BackgroundTransparency=1,Text=string.format("%d/%d",srv.playing,srv.max),
        TextColor3=T.text,TextSize=10,Font=F_MONO,
        TextXAlignment=Enum.TextXAlignment.Left},row)
    mk("TextLabel",{Size=UDim2.new(1,-108,1,0),Position=UDim2.fromOffset(66,0),
        BackgroundTransparency=1,Text=srv.id:sub(1,22),
        TextColor3=T.dim,TextSize=9,Font=F_MONO,
        TextXAlignment=Enum.TextXAlignment.Left},row)
    mk("TextLabel",{Size=UDim2.new(0,42,1,0),Position=UDim2.new(1,-46,0,0),
        BackgroundTransparency=1,Text=tostring(srv.ping),
        TextColor3=(srv.ping<100) and T.text or T.dim,
        TextSize=9,Font=F_MONO,TextXAlignment=Enum.TextXAlignment.Right},row)
    row.MouseButton1Click:Connect(function()
        setStatus("joining...", T.white)
        hopTo(srv.id)
    end)
end

local function renderList()
    for _,ch in ipairs(listFrame:GetChildren()) do
        if ch:IsA("TextButton") then ch:Destroy() end
    end
    if S.filter=="friends" then
        if #S.friends==0 then emptyLabel.Visible=true; emptyLabel.Text="no friends in this game"; return end
        emptyLabel.Visible=false
        for i,f in ipairs(S.friends) do renderFriendRow(f,i) end
        return
    end
    local list=filteredServers()
    if #list==0 then
        emptyLabel.Visible=true
        if S.isLoading then emptyLabel.Text="Loading..."
        elseif S.totalLoaded==0 then emptyLabel.Text="no servers loaded\nhit REFRESH"
        else emptyLabel.Text=string.format("no %s servers\n(%d total)",S.filter=="one" and "1-player" or "open",S.totalLoaded) end
        return
    end
    emptyLabel.Visible=false
    for i,srv in ipairs(list) do renderServerRow(srv,i) end
end

_G.__nullRefresh = function() renderList() end
_G.__nullRenderList = renderList

local function animShow()
    root.Visible = true
    root.Size = UDim2.fromOffset(40, 40)
    root.Position = openerPos

    local info = TweenInfo.new(0.45, Enum.EasingStyle.Cubic, Enum.EasingDirection.Out)
    TweenService:Create(root, info, {
        Size = UDim2.fromOffset(W, H),
        Position = CENTER,
    }):Play()
end
_G.__nullShow = animShow

local function animHide()
    local info = TweenInfo.new(0.4, Enum.EasingStyle.Cubic, Enum.EasingDirection.In)
    local t = TweenService:Create(root, info, {
        Size = UDim2.fromOffset(40, 40),
        Position = openerPos,
    })
    t:Play()
    t.Completed:Connect(function()
        root.Visible = false
        if IS_MOBILE then
            opener.Visible = true
            opener.Size = UDim2.fromOffset(40, 40)
        end
    end)
end

    refreshBtn.MouseButton1Down:Connect(function() pressStart=tick() end)
